@udf.function()
def ejecutar_reporte_financiero() -> str:
    try:
        credential = ClientSecretCredential(
            tenant_id=TENANT_ID,
            client_id=CLIENT_ID,
            client_secret=CLIENT_SECRET
        )

        access_token = credential.get_token(
            "https://api.fabric.microsoft.com/.default"
        ).token

        endpoint = (
            "https://api.fabric.microsoft.com/v1/"
            f"workspaces/{WORKSPACE_ID}/"
            f"dataPipelines/{PIPELINE_ID}/"
            "jobs/execute/instances"
        )

        response = requests.post(
            endpoint,
            headers={
                "Authorization": f"Bearer {access_token}",
                "Content-Type": "application/json"
            },
            timeout=30
        )

        if response.status_code != 202:
            return (
                "No se ha podido iniciar el proceso financiero. "
                f"HTTP {response.status_code}: {response.text[:300]}"
            )

        execution_url = response.headers.get("Location", "")
        job_id = execution_url.rstrip("/").split("/")[-1] \
            if execution_url else "no disponible"

        return (
            "Proceso financiero iniciado correctamente. "
            f"Job: {job_id}"
        )

    except Exception as exc:
        return f"Error inesperado: {str(exc)[:500]}"
